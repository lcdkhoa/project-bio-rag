\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
\label{chapter:ket_luan}

\section{Kết luận}
Đồ án tốt nghiệp ``Nghiên cứu phương pháp RAG đa phương thức cho trợ lý ảo thông minh hỗ trợ dạy và học môn Khoa học tự nhiên cấp Trung học cơ sở'' đã trình bày một cách hệ thống từ cơ sở lý thuyết, kiến trúc hệ thống, quy trình triển khai cho tới thực nghiệm đối chiếu định lượng. Phần này đối chiếu kết quả đạt được với \textbf{năm mục tiêu} đặt ra ở Chương~\ref{chapter:tong_quan_de_tai}, nói rõ mục tiêu nào đã đạt, mục tiêu nào đạt một phần và mục tiêu nào còn nợ.

\begin{enumerate}
    \item \textbf{Mục tiêu 1 --- Truy vấn đa định dạng: đạt một phần.} Hệ thống truy xuất được đồng thời văn bản và hình ảnh, với $3.881$ khung hình được cắt theo neo chú thích có thật trên trang. Riêng vế \textit{công thức}, tại thời điểm các số liệu recall/precision của đồ án được đo, chỉ số dưới vẫn gần như không sống sót qua bước nhận dạng ký tự (tỉ lệ hỏng so với đọc đúng là $256{:}3$, $377{:}3$ và $408{:}4$ trên ba bộ sách) và \textit{văn bản trong kho vẫn hỏng}. Một biện pháp \textit{một phần} đang chạy trong sản xuất --- chuẩn hoá công thức ở phía truy vấn và chỉ mục thưa, không sửa văn bản đã lưu --- nâng số truy vấn tìm được đoạn đúng từ $1/12$ lên $11/12$ trên một mẫu $12$ công thức. Biện pháp triệt để hơn --- kiến trúc lai OCR chỉ gọi mô hình thị giác-ngôn ngữ cho vùng công thức, giữ Tesseract cho văn xuôi (Mục~\ref{sec:formula_hybrid}) --- \textbf{đã thiết kế, cài đặt và kiểm thử xong} nhưng chưa chạy trên corpus sản xuất; mục tiêu này vì vậy vẫn ở trạng thái \textit{đạt một phần} cho tới lượt lập chỉ mục lại kế tiếp.

    \item \textbf{Mục tiêu 2 --- Kho vector toàn diện: đạt.} Quy trình ETL hai luồng đã số hoá và lập chỉ mục \textbf{2.387/2.399 trang} của đủ 12 cuốn sách giáo khoa Khoa học tự nhiên lớp 6--9 thuộc ba bộ sách, tích hợp cả ba phân môn Vật lí, Hoá học và Sinh học --- chênh lệch đúng $12$ trang là trang bìa, được bỏ theo thiết kế. Kết quả là \textbf{20.274 vector} ($16.393$ vector văn bản $1024$ chiều và $3.881$ vector hình ảnh $512$ chiều), kèm một chỉ mục thưa BM25 dùng chung định danh đoạn. Toàn bộ quá trình chạy trên CPU của một máy trạm thông thường.

    \item \textbf{Mục tiêu 3 --- Tối ưu pha truy xuất: đạt, và đây là đóng góp phương pháp chính.} Khoảng cách recall mà đề cương nêu làm trọng tâm đã bị xoá: recall của cấu hình thật là $\mathbf{0{,}9091}$, \textit{cao hơn} $0{,}8961$ của top-10 thô đo trên riêng kênh ngữ nghĩa. Hai biện pháp có tác dụng, cả hai đều có số trước/sau: \textbf{tắt} cổng lọc theo khoảng cách tương đối (R@10 của kênh lai $0{,}892 \rightarrow 0{,}970$) và \textbf{bật} xếp hạng lại bằng cross-encoder (R@1 của kênh ngữ nghĩa $0{,}602 \rightarrow 0{,}732$, tức $+21{,}6\%$). Điều đáng ghi nhận về mặt phương pháp là biện pháp được kỳ vọng nhiều nhất --- siết nhiễu bằng cổng lọc --- hoá ra \textit{có hại}, và chỉ có phép đo đối chiếu mới phát hiện được điều đó.

    \item \textbf{Mục tiêu 4 --- Khung đánh giá đối chiếu: đạt vế thứ nhất, chưa kết luận được vế thứ hai.} Bộ kiểm thử $231$ câu có gắn nhãn nguồn, phủ đủ 12 cuốn, cho phép đo cả hai pha. Vế \textit{truy xuất lai so với BM25 thuần} kèm nghiên cứu loại bỏ thành phần đã hoàn tất: ở đúng bề rộng production, truy xuất lai thắng ở \textit{mọi} độ đo (R@1 $0{,}740$ · R@3 $0{,}909$ · R@10 $0{,}970$ · MRR $0{,}826$). Vế \textit{đa phương thức so với chỉ văn bản} cho chênh lệch $+0{,}010$ recall, tức đúng một câu hỏi --- nhưng kết luận đúng là \textbf{``chưa đo được ưu thế''} chứ không phải ``không có ưu thế'', vì đáp án chuẩn của bộ kiểm thử được sinh từ chính văn bản trang vàng nên dư địa còn lại cho kênh hình chỉ là $0{,}104$.

    \item \textbf{Mục tiêu 5 --- Giao diện web: đạt.} Ứng dụng Next.js cho phép đặt câu hỏi, nhận câu trả lời theo luồng, xem hình minh hoạ do máy chủ phục vụ, và --- sau lượt bổ sung cuối --- \textbf{dựng công thức đúng ký hiệu bằng KaTeX} cùng \textbf{hiển thị danh sách trích dẫn} thành một phần riêng. Trước đó cả hai yêu cầu này đều chưa đáp ứng; riêng phần trích dẫn thì lỗi nằm ở \textit{cả hai} phía chứ không chỉ ở giao diện, vì máy chủ mới chỉ chèn khối nguồn dạng chữ vào giữa câu trả lời chứ chưa trả nó thành một trường dữ liệu (Mục~\ref{sec:fe_mt5}).
\end{enumerate}

Về chất lượng câu trả lời, mô hình giám khảo độc lập chấm trên $231$ câu cho Độ liên quan $4{,}602/5$, Độ trung thực $4{,}394/5$ và Tính đúng $4{,}065/5$. Điểm Độ trung thực cao khẳng định hiệu quả của cơ chế RAG trong việc buộc mô hình bám sát sách giáo khoa chính thống; việc Tính đúng thấp hơn hai tiêu chí kia ở \textit{cả 12 cuốn} cho thấy phần cải thiện còn lại vẫn nằm ở pha truy xuất chứ không ở khả năng diễn đạt.

Một kết quả không nằm trong danh sách mục tiêu nhưng đáng ghi lại: \textbf{trích dẫn của hệ thống là xác định, không do mô hình sinh}. Số trang được dựng từ siêu dữ liệu của chính đoạn văn bản được truy xuất, nên mô hình ngôn ngữ \textit{không có khả năng} bịa ra một số trang, kể cả khi nó bịa nội dung. Với một hệ thống phục vụ học sinh phổ thông, đây là ranh giới quan trọng hơn bất kỳ điểm số nào.

\section{Hạn chế của đề tài}
Các hạn chế dưới đây đều \textit{đã đo được}, kể cả những hạn chế bất lợi cho kết luận của đồ án; chi tiết số liệu nằm ở Mục~\ref{sec:hanche}.

\begin{itemize}
    \item \textbf{Bộ kiểm thử do mô hình sinh và chưa được người duyệt toàn bộ.} Mẫu $50$ câu người rà soát cho tỉ lệ nhãn nguồn sai $4{,}1\%$, khoảng tin cậy Wilson $95\%$ là $1{,}1\%$--$13{,}7\%$; cận trên nghĩa là tới khoảng $9/231$ câu vẫn có thể sai. Riêng $39$ câu hỏi hình thì \textbf{chưa ai đối chiếu với ảnh} --- mô hình viết chúng từ chú thích và chữ nhận dạng, không nhìn thấy hình.
    \item \textbf{Bước xử lý đặc thù cho công thức Hoá, Lí đã cài đặt xong nhưng chưa chạy trên corpus sản xuất} (Mục tiêu 1). Một cuộc đối chiếu bốn công cụ nhận dạng cho thấy \textit{có} công cụ đọc đúng \ce{CO2} và \ce{O2}, nhưng công cụ đó bị loại khỏi vai trò thay thế toàn bộ vì tỉ lệ lỗi dấu tiếng Việt cao gấp hơn hai lần công cụ đang dùng ($0{,}037$ so với $0{,}016$). Kiến trúc lai chỉ gọi công cụ đó cho dòng bị nghi có công thức hỏng (Mục~\ref{sec:formula_hybrid}) đã hoàn thành cài đặt và kiểm thử, còn thiếu duy nhất một lượt chạy trên môi trường GPU để lập chỉ mục lại toàn bộ $12$ cuốn.
    \item \textbf{Số hiệu Bài chỉ có ở $4/12$ cuốn} ($4.857/16.393$ đoạn), vì tám cuốn còn lại chưa đọc được trục Bài liền mạch và hệ thống cố tình để trống thay vì đoán.
    \item \textbf{$11.459$ đoạn của hai bộ sách vẫn mang nhãn nhà xuất bản sai trong siêu dữ liệu.} Lỗi ở mã nguồn đã sửa, nhưng dữ liệu chỉ sạch sau một lượt lập chỉ mục lại toàn bộ, nên tới thời điểm viết báo cáo nhãn này \textbf{vẫn sai}.
    \item \textbf{Khung cắt hình ghép còn hụt.} Tỉ lệ khung cắt dạng dải dọc hẹp đã giảm từ $17{,}5\%$ xuống $1{,}7\%$ ở một bộ sách, nhưng ca gốc --- một hình ghép ba tấm --- \textbf{vẫn cắt sai}; phần hụt còn lại chưa lượng hoá được.
    \item \textbf{Cờ ``cần rà soát'' mất tác dụng phân biệt}, vì bật ở $69{,}3\%$ số đoạn.
    \item \textbf{Đoạn văn bản không giữ quan hệ hàng--cột của bảng}, nên hệ thống có thể trả lời bằng dữ liệu của sai cột.
    \item \textbf{Giao diện mới chỉ được kiểm bằng kịch bản tự động trên máy cục bộ}; chưa có học sinh hay giáo viên nào dùng thử.
\end{itemize}

\section{Hướng phát triển tương lai}
Thứ tự dưới đây được sắp theo \textit{mức đóng góp cho các mục tiêu còn nợ}, không theo mức độ thú vị về kỹ thuật.

\begin{itemize}
    \item \textbf{Đưa giao diện ra thử nghiệm thật.} Ba yêu cầu của Mục tiêu 5 đã xong về mặt chức năng, nhưng toàn bộ mới chỉ được kiểm bằng kịch bản tự động trên máy cục bộ. Bước còn thiếu là người dùng thật, không phải thêm tính năng.
    \item \textbf{Chạy kiến trúc lai xử lý công thức Hoá, Lí trên corpus sản xuất --- Mục tiêu 1:} phần thiết kế, cài đặt và kiểm thử đã xong (Mục~\ref{sec:formula_hybrid}) --- dùng công cụ nhận dạng đọc được chỉ số dưới \textit{chỉ cho vùng công thức}, giữ công cụ hiện tại cho văn xuôi, để không đánh đổi độ chính xác dấu tiếng Việt của toàn bộ kho lấy vài công thức. Việc còn lại là chạy lượt lập chỉ mục lại trên môi trường GPU rồi đo lại Recall/MRR/Precision để xác nhận không có hồi quy. Nguyên tắc bắt buộc giữ nguyên: \textbf{không bao giờ viết lại văn bản đã nhận dạng để ``chữa'' công thức}, vì đoán lại một chỉ số dưới chính là bịa dữ liệu.
    \item \textbf{Bộ kiểm thử sinh từ hình, có người đối chiếu ảnh.} Đây là điều kiện \textit{tiên quyết} để kết luận được vế thứ hai của Mục tiêu 4: chừng nào đáp án chuẩn còn sinh từ văn bản trang vàng thì dư địa cho kênh hình còn bị chặn ở $0{,}104$, và mọi so sánh đa phương thức đều đo trong một cái trần quá thấp.
    \item \textbf{Hoàn thiện trục Bài cho tám cuốn còn lại}, qua đó mở khoá truy vấn theo Bài trên toàn kho. Việc này không đòi lập chỉ mục lại: chỉ cần sửa bộ đọc mục lục cho hai bố cục còn lại rồi nạp bù siêu dữ liệu.
    \item \textbf{Giữ cấu trúc hàng--cột của bảng số liệu} khi phân mảnh, để loại bỏ hẳn lớp lỗi ``trả lời đúng nội dung nhưng sai cột''.
    \item \textbf{Nới số đoạn giữ lại và đo lại chất lượng câu trả lời.} Phép đo hiện tại cho biết chính xác cái giá của việc cắt còn ba đoạn: $0{,}061$ recall. Nới lên mười đoạn sẽ thu lại đúng phần đó, đổi lại ngữ cảnh dài gấp ba --- một đánh đổi \textit{định lượng được} nhưng chưa được đo về phía Tính đúng của câu trả lời.
    \item \textbf{Lượng tử hoá mô hình sinh} (GGUF, AWQ, GPTQ) để hạ chi phí phục vụ. Cần nói rõ đây là bài toán của pha \textit{phục vụ}: pha ETL đã chạy trọn trên CPU, nên nút thắt phần cứng không nằm ở khâu xây dựng kho tri thức.
    \item \textbf{Triển khai thử nghiệm và lấy ý kiến người dạy.} Toàn bộ đánh giá trong đồ án này là đánh giá tự động; chưa có giáo viên bộ môn nào thẩm định câu trả lời của hệ thống. Đây là bước xác nhận còn thiếu trước khi nói tới việc dùng thật trong lớp học.  % SO-CU-CO-Y
\end{itemize}
