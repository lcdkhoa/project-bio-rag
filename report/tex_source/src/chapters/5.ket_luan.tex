\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
\label{chapter:ket_luan}

\section{Kết luận}
Đồ án tốt nghiệp ``Nghiên cứu phương pháp RAG đa phương thức cho trợ lý ảo thông minh hỗ trợ dạy và học môn Khoa học tự nhiên cấp Trung học cơ sở'' đã trình bày một cách hệ thống từ cơ sở lý thuyết, kiến trúc hệ thống, quy trình triển khai cho tới thực nghiệm đối chiếu định lượng. Phần này đối chiếu kết quả đạt được với \textbf{năm mục tiêu} đặt ra ở Chương~\ref{chapter:tong_quan_de_tai}, nói rõ mục tiêu nào đã đạt, mục tiêu nào đạt một phần và mục tiêu nào còn nợ.

\begin{enumerate}
    \item \textbf{Mục tiêu 1 --- Truy vấn đa định dạng: đạt một phần.} Hệ thống truy xuất được đồng thời văn bản và hình ảnh, với $3.881$ khung hình được cắt theo neo chú thích có thật trên trang. Riêng vế \textit{công thức}, bake-off trên gold set $97$ ô đo được chỉ số dưới gần như không sống sót qua bước nhận dạng ký tự (tỉ lệ hỏng so với đọc đúng là $256{:}3$, $377{:}3$ và $408{:}4$ trên ba bộ sách). Một biện pháp \textit{một phần} đang chạy trong sản xuất --- chuẩn hoá công thức ở phía truy vấn và chỉ mục thưa, không sửa văn bản đã lưu --- nâng số truy vấn tìm được đoạn đúng từ $1/12$ lên $11/12$ trên một mẫu $12$ công thức. Biện pháp triệt để hơn --- kiến trúc lai OCR chỉ gọi mô hình thị giác-ngôn ngữ cho vùng công thức, giữ Tesseract cho văn xuôi (Mục~\ref{sec:formula_hybrid}) --- \textbf{đã chạy trên toàn bộ corpus sản xuất}: $4\,293$ lượt merge MinerU thành công trên $16\,515$ đoạn, $0$ lỗi gọi mô hình. Mục tiêu vẫn ở trạng thái \textit{đạt một phần} vì mức cải thiện thật của chỉ số dưới sau khi merge \textbf{chưa được đo lại} bằng cùng bake-off đã dùng để phát hiện vấn đề --- đây là việc còn lại duy nhất.

    \item \textbf{Mục tiêu 2 --- Kho vector toàn diện: đạt.} Quy trình ETL hai luồng đã số hoá và lập chỉ mục \textbf{2.387/2.399 trang} của đủ 12 cuốn sách giáo khoa Khoa học tự nhiên lớp 6--9 thuộc ba bộ sách, tích hợp cả ba phân môn Vật lí, Hoá học và Sinh học --- chênh lệch đúng $12$ trang là trang bìa, được bỏ theo thiết kế. Kết quả là \textbf{20.396 vector} ($16.515$ vector văn bản $1024$ chiều và $3.881$ vector hình ảnh $512$ chiều), kèm một chỉ mục thưa BM25 dùng chung định danh đoạn. Toàn bộ quá trình chạy trên CPU của một máy trạm thông thường.

    \item \textbf{Mục tiêu 3 --- Tối ưu pha truy xuất: đạt phần lớn, và đây là đóng góp phương pháp chính.} Đo lại trên bộ kiểm thử $240$ câu lấy mẫu ngẫu nhiên (Mục~\ref{sec:testset}), ở bề rộng production ($20$ ứng viên/kênh, $210$ câu có nhãn nguồn): cấu hình đề xuất (truy xuất lai $+$ xếp hạng lại, cổng lọc tắt) đạt \textbf{MRR $0{,}7789$}, thắng cả ba cấu hình đối chứng ở MRR/R@1/R@3/R@5 (BM25 thuần $0{,}6636$; ngữ nghĩa thuần $0{,}5664$; lai chưa xếp hạng lại $0{,}6947$). Một ngoại lệ đáng ghi: R@10 của cấu hình đề xuất ($0{,}2458$) thấp hơn cấu hình lai chưa xếp hạng lại ($0{,}2643$) --- một nghịch lý còn là \textbf{câu hỏi mở}, giả thuyết hợp lý nhất là ngưỡng lọc điểm xếp hạng lại (\texttt{RERANK\_SCORE\_MIN=0,59}) cắt bớt một số ứng viên đúng, nhưng chưa có phép đo riêng xác nhận. Hai biện pháp có tác dụng nhất quán với mọi lượt đo trước: \textbf{tắt} cổng lọc theo khoảng cách tương đối (MRR cấu hình đề xuất $0{,}7157 \rightarrow 0{,}7779$ khi tắt) và \textbf{bật} xếp hạng lại bằng cross-encoder (R@1 kênh ngữ nghĩa $0{,}0647 \rightarrow 0{,}0994$, tức $+53{,}6\%$, mức cải thiện lớn nhất trong ba kênh). Đơn vị đo lần này là tỉ lệ \textit{chuẩn} các đoạn của trang vàng lọt vào top-$k$ (không phải hit@k nhị phân của các lượt trước), nên giá trị tuyệt đối không so sánh trực tiếp được với các con số recall mức-trang đã công bố trước đây; điều so sánh được và giữ nguyên qua mọi lượt đo là \textbf{thứ hạng giữa các cấu hình} và \textbf{hướng tác dụng của từng công tắc}.

    \item \textbf{Mục tiêu 4 --- Khung đánh giá đối chiếu: đạt cả ba vế.} Bộ kiểm thử $240$ câu có gắn nhãn nguồn (trừ $30$ câu ngoài phạm vi không có nhãn theo thiết kế), lấy mẫu ngẫu nhiên trên toàn corpus, cho phép đo cả hai pha. Vế \textit{truy xuất lai so với BM25 thuần} kèm nghiên cứu loại bỏ thành phần đã hoàn tất (số liệu ở Mục tiêu 3). Vế \textit{đa phương thức so với chỉ văn bản} nay có \textbf{kết luận dứt khoát} (D-189), khác hẳn trạng thái ``chưa đo được ưu thế'' của các lượt trước: đo trên đúng $52$ câu hình có \texttt{ground\_truth} sinh từ chính siêu dữ liệu của hình (không còn bị trần $0{,}104$ của bộ câu cũ chặn lại), ngữ cảnh đa phương thức \textbf{không cải thiện, tệ đi nhẹ} ở cả ba tiêu chí giám khảo (Tính đúng $\Delta{-0{,}077}$; Độ trung thực và Độ liên quan $\Delta{-0{,}038}$ mỗi tiêu chí), nên hệ thống giữ \texttt{MULTIMODAL\_CONTEXT\_ENABLED=false}. Vế thứ ba, mới xuất hiện ở lượt đo này: khả năng \textbf{từ chối đúng khi câu hỏi ngoài phạm vi} --- $29/30$ câu ($0{,}9667$), điểm giám khảo gần tuyệt đối ($4{,}87$/$5{,}00$/$5{,}00$) --- một phép đo trực tiếp cho nguyên tắc không bịa mà các lượt đo trước chưa từng có.  % SO-CU-CO-Y

    \item \textbf{Mục tiêu 5 --- Giao diện web: đạt.} Ứng dụng Next.js cho phép đặt câu hỏi, nhận câu trả lời theo luồng, xem hình minh hoạ do máy chủ phục vụ, và --- sau lượt bổ sung cuối --- \textbf{dựng công thức đúng ký hiệu bằng KaTeX} cùng \textbf{hiển thị danh sách trích dẫn} thành một phần riêng. Trước đó cả hai yêu cầu này đều chưa đáp ứng; riêng phần trích dẫn thì lỗi nằm ở \textit{cả hai} phía chứ không chỉ ở giao diện, vì máy chủ mới chỉ chèn khối nguồn dạng chữ vào giữa câu trả lời chứ chưa trả nó thành một trường dữ liệu (Mục~\ref{sec:fe_mt5}).
\end{enumerate}

Về chất lượng câu trả lời, một tập bốn mô hình giám khảo xoay vòng (Groq) chấm độc lập trên $240$ câu, tách theo LOẠI câu hỏi: nhóm \textbf{ngoài phạm vi} gần như tuyệt đối (Tính đúng $4{,}87/5$, Độ trung thực và Độ liên quan $5{,}00/5$) --- bằng chứng trực tiếp hệ thống từ chối đúng thay vì bịa; nhóm \textbf{văn bản} đạt mức khá (Tính đúng $4{,}19/5$, Độ trung thực $4{,}56/5$, Độ liên quan $4{,}59/5$); nhóm \textbf{hình} yếu nhất (Tính đúng $3{,}08/5$, Độ trung thực $3{,}65/5$, Độ liên quan $3{,}90/5$). Ở cả ba nhóm, Tính đúng luôn là tiêu chí thấp nhất, cho thấy phần cải thiện còn lại vẫn nằm ở pha truy xuất chứ không ở khả năng diễn đạt.

Một kết quả không nằm trong danh sách mục tiêu nhưng đáng ghi lại: \textbf{trích dẫn của hệ thống là xác định, không do mô hình sinh}. Số trang được dựng từ siêu dữ liệu của chính đoạn văn bản được truy xuất, nên mô hình ngôn ngữ \textit{không có khả năng} bịa ra một số trang, kể cả khi nó bịa nội dung. Với một hệ thống phục vụ học sinh phổ thông, đây là ranh giới quan trọng hơn bất kỳ điểm số nào.

\section{Hạn chế của đề tài}
Các hạn chế dưới đây đều \textit{đã đo được}, kể cả những hạn chế bất lợi cho kết luận của đồ án; chi tiết số liệu nằm ở Mục~\ref{sec:hanche}.

\begin{itemize}
    \item \textbf{Bộ kiểm thử do mô hình sinh, duyệt bằng một lượt đọc toàn file duy nhất --- không còn mẫu độc lập đo tỉ lệ lỗi còn sót lại.} Tỉ lệ nhãn nguồn sai $4{,}1\%$ đo được ở lượt trước (mẫu $50$ câu) là số đo cho một bộ câu hỏi khác, sinh bằng thuật toán khác; áp nó cho bộ $240$ câu hiện tại là gán nhầm bằng chứng nên đồ án không làm vậy --- đây là một khoảng trống đo lường thật, chưa có số thay thế.
    \item \textbf{Mô hình giám khảo không đồng nhất giữa các câu.} Việc mở rộng từ hai lên bốn mô hình giám khảo (Groq, để tránh hạn mức $200\,000$ token/ngày/mô hình) khiến các câu khác nhau có thể được chấm bởi các mô hình khác nhau trong tập bốn mô hình --- một nguồn nhiễu chưa đo tách riêng được ảnh hưởng.
    \item \textbf{Bước xử lý đặc thù cho công thức Hoá, Lí đã chạy trên corpus sản xuất, nhưng mức cải thiện chưa được đo lại} (Mục tiêu 1). Một cuộc đối chiếu bốn công cụ nhận dạng cho thấy \textit{có} công cụ đọc đúng \ce{CO2} và \ce{O2}, nhưng công cụ đó bị loại khỏi vai trò thay thế toàn bộ vì tỉ lệ lỗi dấu tiếng Việt cao gấp hơn hai lần công cụ đang dùng ($0{,}037$ so với $0{,}016$). Kiến trúc lai chỉ gọi công cụ đó cho dòng bị nghi có công thức hỏng (Mục~\ref{sec:formula_hybrid}) đã chạy trên toàn bộ $12$ cuốn ($4\,293$ lượt merge thành công); còn thiếu một lượt bake-off mới để lượng hoá mức cải thiện thật của chỉ số dưới trên văn bản đã merge.
    \item \textbf{Số hiệu Bài chỉ có ở $4/12$ cuốn} ($4.919/16.515$ đoạn, $29{,}8\%$), vì tám cuốn còn lại chưa đọc được trục Bài liền mạch và hệ thống cố tình để trống thay vì đoán.
    \item \textbf{Trục tổng hợp đổi sang theo LOẠI câu hỏi làm mất bảng xếp hạng $12$ cuốn sách của phiên bản trước.} Đây là đánh đổi có chủ đích theo chỉ đạo CBHD, nhưng đồ án hiện không còn một góc nhìn định lượng riêng cho từng cuốn sách.
    \item \textbf{Khung cắt hình ghép còn hụt.} Tỉ lệ khung cắt dạng dải dọc hẹp đã giảm từ $17{,}5\%$ xuống $1{,}7\%$ ở một bộ sách, nhưng ca gốc --- một hình ghép ba tấm --- \textbf{vẫn cắt sai}; phần hụt còn lại chưa lượng hoá được.
    \item \textbf{Cờ ``cần rà soát'' mất tác dụng phân biệt}, vì bật ở $70{,}8\%$ số đoạn.
    \item \textbf{R@10 của cấu hình đề xuất thấp hơn cấu hình chưa xếp hạng lại} --- một nghịch lý đo được nhưng chưa lý giải chắc chắn (Mục tiêu 3).
    \item \textbf{Đoạn văn bản không giữ quan hệ hàng--cột của bảng}, nên hệ thống có thể trả lời bằng dữ liệu của sai cột.
    \item \textbf{Giao diện mới chỉ được kiểm bằng kịch bản tự động trên máy cục bộ}; chưa có học sinh hay giáo viên nào dùng thử.
\end{itemize}

\section{Hướng phát triển tương lai}
Thứ tự dưới đây được sắp theo \textit{mức đóng góp cho các mục tiêu còn nợ}, không theo mức độ thú vị về kỹ thuật.

\begin{itemize}
    \item \textbf{Đưa giao diện ra thử nghiệm thật.} Ba yêu cầu của Mục tiêu 5 đã xong về mặt chức năng, nhưng toàn bộ mới chỉ được kiểm bằng kịch bản tự động trên máy cục bộ. Bước còn thiếu là người dùng thật, không phải thêm tính năng.
    \item \textbf{Đo lại bake-off công thức Hoá, Lí sau khi kiến trúc lai đã chạy trên corpus sản xuất --- Mục tiêu 1:} phần chạy sản xuất đã xong (Mục~\ref{sec:formula_hybrid}, $4\,293$ lượt merge thành công trên $16\,515$ đoạn) --- dùng công cụ nhận dạng đọc được chỉ số dưới \textit{chỉ cho vùng công thức}, giữ công cụ hiện tại cho văn xuôi, để không đánh đổi độ chính xác dấu tiếng Việt của toàn bộ kho lấy vài công thức. Việc còn lại là chạy lại đúng bộ bake-off ($97$ ô/$15$ trang) trên văn bản đã merge để xác nhận tỉ lệ hỏng:đúng thật sự cải thiện, rồi đo lại Recall/MRR/Precision để xác nhận không có hồi quy. Nguyên tắc bắt buộc giữ nguyên: \textbf{không bao giờ viết lại văn bản đã nhận dạng để ``chữa'' công thức}, vì đoán lại một chỉ số dưới chính là bịa dữ liệu.
    \item \textbf{Điều tra nghịch lý R@10 của cấu hình đề xuất.} Mục tiêu 3 đo được cấu hình đề xuất thua nhẹ cấu hình lai chưa xếp hạng lại ở R@10 dù thắng ở mọi cột khác; giả thuyết \texttt{RERANK\_SCORE\_MIN} lọc bỏ ứng viên đúng cần một phép đo quét ngưỡng để xác nhận hoặc bác bỏ, thay vì dừng lại ở suy đoán.
    \item \textbf{Bổ sung một mẫu độc lập thứ hai để đo tỉ lệ lỗi nhãn nguồn của bộ $240$ câu hiện tại.} Quy trình duyệt hiện tại là một lượt đọc toàn file, không có cơ chế kiểm chéo như bộ câu hỏi hình của quy trình cũ; một mẫu con được người thứ hai rà soát độc lập sẽ cho một khoảng tin cậy thay cho việc để trống hoàn toàn.
    \item \textbf{Hoàn thiện trục Bài cho tám cuốn còn lại}, qua đó mở khoá truy vấn theo Bài trên toàn kho. Việc này không đòi lập chỉ mục lại: chỉ cần sửa bộ đọc mục lục cho hai bố cục còn lại rồi nạp bù siêu dữ liệu.
    \item \textbf{Giữ cấu trúc hàng--cột của bảng số liệu} khi phân mảnh, để loại bỏ hẳn lớp lỗi ``trả lời đúng nội dung nhưng sai cột''.
    \item \textbf{Đo lại chất lượng câu trả lời khi nới số đoạn giữ lại.} R@10 của cấu hình đề xuất ($0{,}2458$) cao hơn R@5 ($0{,}2085$), cho thấy vẫn còn recall thu được nếu nới ngữ cảnh; đây là một đánh đổi định lượng được ở phía truy xuất nhưng chưa đo về phía Tính đúng của câu trả lời.
    \item \textbf{Lượng tử hoá mô hình sinh} (GGUF, AWQ, GPTQ) để hạ chi phí phục vụ. Cần nói rõ đây là bài toán của pha \textit{phục vụ}: pha ETL đã chạy trọn trên CPU, nên nút thắt phần cứng không nằm ở khâu xây dựng kho tri thức.
    \item \textbf{Triển khai thử nghiệm và lấy ý kiến người dạy.} Toàn bộ đánh giá trong đồ án này là đánh giá tự động; chưa có giáo viên bộ môn nào thẩm định câu trả lời của hệ thống. Đây là bước xác nhận còn thiếu trước khi nói tới việc dùng thật trong lớp học.  % SO-CU-CO-Y
\end{itemize}
